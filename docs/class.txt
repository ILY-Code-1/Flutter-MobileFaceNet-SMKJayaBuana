class AdminAccount {
  - id : int
  - username : String
  - pinHash : String
  - createdAt : DateTime
}

class SchoolClass {
  - id : int
  - name : String
  - createdAt : DateTime
}

class Student {
  - id : int
  - name : String
  - nis : String
  - classId : int
  - photoPath : String
  - embedding : List<double>
  - isDraft : bool
  - createdAt : DateTime
  + initials() : String
  + hue() : int
}

class AttendanceRecord {
  - id : int
  - studentId : int
  - date : String
  - checkInTime : String
  - checkOutTime : String
  - status : AttendanceStatus
  - autoCheckout : bool
}

enum AttendanceStatus {
  present
  late
  absent
  + code() : String
  + label() : String
}

class AppSettings {
  - activeClassIds : List<int>
  - clockIn : String
  - clockOut : String
  - lastCheckOut : String
  - deviceName : String
  - soundOnScan : String
}
