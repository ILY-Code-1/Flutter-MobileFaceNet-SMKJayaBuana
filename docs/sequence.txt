' == Sequence: Registrasi Akun Admin ==
@startuml
title Sequence: Registrasi Akun Admin
actor Admin
participant "RegistrationPage" as UI
participant "AppDatabase" as DB
participant "DbSeeder" as Seed

Admin -> UI : Isi username & PIN 6 digit
activate UI
UI -> UI : Validasi PIN (PinValidator)
UI -> DB : createAdmin(username, pin)
activate DB
DB --> UI : akun admin tersimpan
deactivate DB
UI -> Seed : seedIfEmpty()
activate Seed
Seed -> DB : insert kelas & siswa awal
Seed --> UI : data awal selesai dibuat
deactivate Seed
UI --> Admin : Buka layar kamera absensi
deactivate UI
@enduml

' == Sequence: Melakukan Absensi Wajah ==
@startuml
title Sequence: Melakukan Absensi Wajah
actor Siswa
participant "CameraIdlePage" as UI
participant "FaceRecognitionService" as FR
participant "AppDatabase" as DB

Siswa -> UI : Tekan "Mulai Scan"
activate UI
UI -> FR : identifyFromFile(foto, kelasAktif)
activate FR
FR -> DB : listStudents(kelas aktif)
DB --> FR : daftar siswa + embedding
FR --> UI : FaceMatch (siswa, similarity)
deactivate FR
alt Wajah dikenali
  UI -> Siswa : Tampilkan data & pilihan absensi
  Siswa -> UI : Pilih Check-in / Check-out
  UI -> DB : upsertCheckIn / updateCheckOut
  DB --> UI : data absensi tersimpan
  UI --> Siswa : Tampilkan halaman sukses
else Tidak dikenali
  UI --> Siswa : Tampilkan "Wajah tidak dikenali"
end
deactivate UI
@enduml

' == Sequence: Hubungi Developer & Backup Database ==
@startuml
title Sequence: Hubungi Developer & Backup Database
actor Admin
participant "MenuPage" as UI
participant "DatabaseBackupService" as Backup
participant "AppDatabase" as DB

alt Contact developer
  Admin -> UI : Tekan "Contact developer"
  activate UI
  UI --> Admin : Buka WhatsApp developer berisi pesan keluhan
  deactivate UI
else Backup Database
  Admin -> UI : Tekan "Backup Database"
  activate UI
  UI -> Backup : writeSqlBackup()
  activate Backup
  Backup -> DB : exportSqlDump()
  activate DB
  DB --> Backup : skrip SQL basis data
  deactivate DB
  Backup --> UI : berkas .sql cadangan
  deactivate Backup
  UI -> UI : Buka lembar berbagi berkas
  UI --> Admin : Kirim berkas cadangan ke developer
  deactivate UI
end
@enduml
